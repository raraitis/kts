💡
Kas ir API? API (Application Programming Interface) ir veids, kā viena programma var lūgt datus no citas. Piemēram, mūsu mājaslapa lūdz uzņēmumu datus no data.gov.lv un parāda tos saprotamā veidā.
📖
GET nozīmē, ka mēs lasām datus. Parametri tiek pievienoti URL aiz ? zīmes. Šis pieprasījums veic pilna teksta meklēšanu — atrod uzņēmumus, kuros jebkurā laukā ir ievadītais vārds.
📖

export interface Company {
  _id:               number;        // iekšējais DB ID
  regcode:           string;        // reģistrācijas numurs (11 cipari)
  sepa:              string;        // SEPA maksājuma identifikators
  name:              string;        // pilns nosaukums
  name_before_quotes:string;        // nosaukuma daļa pirms pēdiņām
  name_in_quotes:    string;        // nosaukuma daļa pēdiņās
  name_after_quotes: string;        // nosaukuma daļa pēc pēdiņām
  without_quotes:    number;        // 1 ja nosaukumā nav pēdiņu
  regtype:           string;        // reģistra veida kods  (U, B, ...)
  regtype_text:      string;        // reģistra veida nosaukums
  type:              string;        // juridiskā forma (SIA, AS, ...)
  type_text:         string;        // juridiskās formas pilns nosaukums
  registered:        string;        // ISO datums — reģistrācijas datums
  terminated:        string | null; // ISO datums — izslēgšanas datums
  closed:            string;        // ' ' = aktīvs, 'L' = likvidēts, ...
  address:           string;        // juridiskā adrese
  index:             number;        // pasta indekss
  addressid:         number;        // adreses reģistra ID
  region:            number;        // reģiona kods
  city:              number;        // pilsētas / novada kods
  atvk:              number;        // ATVK teritorijas kods
  reregistration_term: string;      // pārreģistrācijas termiņš
}

/** Atbilde, ko atgriež /api/search */
export interface SearchResponse {
  total:   number;    // kopējais atbilstošo ierakstu skaits datubāzē
  records: Company[]; // šajā lapā atgrieztie ieraksti
  meta: {
    resource_id:  string; // CKAN resursa ID
    limit:        number; // cik ierakstu prasīti
    offset:       number; // no kura ieraksta sākts
    upstream_url: string; // faktiskais URL uz data.gov.lv
  };
}
    
        
    
      
    

      
      
    
        
    Meklēt pēc nosaukuma
    
        
    Izmanto fetch ar GET metodi. Parametrs q meklē visās teksta kolonnās.
        Opcijas limit un offset ļauj veidot lapošanu.
    
        
    
          
    
            TypeScript
            
          
    
          
    import type { SearchResponse } from './types'; // iekopē tipus no augšas

/**
 * Meklē uzņēmumus pēc nosaukuma vai jebkura teksta lauka.
 *
 * @param query  - meklēšanas frāze, piem. "SIA Latvijas"
 * @param limit  - cik rezultātus atgriezt (max 50, noklusējums 10)
 * @param offset - izlaist N ierakstus (lapošanai)
 * @returns      - masīvs ar atrastajiem uzņēmumiem
 */
async function searchByName(
  query: string,
  limit = 10,
  offset = 0
): Promise<SearchResponse> {
  // Veidojam URL ar parametriem
  const url = new URL('http://localhost:3000/api/search');
  url.searchParams.set('q',      query);
  url.searchParams.set('limit',  String(limit));
  url.searchParams.set('offset', String(offset));

  const response = await fetch(url.toString());

  // Ja serveris atgrieza kļūdas statusu, izmetam kļūdu
  if (!response.ok) {
    const err = await response.json() as { error: string };
    throw new Error(err.error);
  }

  return response.json() as Promise<SearchResponse>;
}

// ── Lietošanas piemērs ────────────────────────────────────────────────────
const result = await searchByName('Latvijas Banka', 5);
console.log(`Atrasti: ${result.total}`);
result.records.forEach(c => {
  console.log(`${c.regcode}  ${c.name}`);
});
    
        
    
      
    

      
      
    
        
    Meklēt pēc reģistrācijas numura
    
        
    Parametrs regcode veic precīzu meklēšanu — atgriež tieši to uzņēmumu,
        kuram ir šis numurs. Noderīgi, ja numurs jau ir zināms.
    
        
    
          
    
            TypeScript
            
          
    
          
    import type { SearchResponse, Company } from './types';

/**
 * Atrod uzņēmumu pēc precīza reģistrācijas numura.
 *
 * @param regcode - 11 ciparu reģistrācijas numurs, piem. "40003009497"
 * @returns         Company objekts vai null, ja netika atrasts
 */
async function findByRegcode(regcode: string): Promise<Company | null> {
  const url = new URL('http://localhost:3000/api/search');
  url.searchParams.set('regcode', regcode);

  const response = await fetch(url.toString());

  if (!response.ok) {
    const err = await response.json() as { error: string };
    throw new Error(err.error);
  }

  const data = await response.json() as SearchResponse;

  // Atgriežam pirmo ierakstu, vai null ja nekas netika atrasts
  return data.records[0] ?? null;
}

// ── Lietošanas piemērs ────────────────────────────────────────────────────
const company = await findByRegcode('40003009497');
if (company) {
  console.log(company.name);    // "LATVIJAS BANKA"
  console.log(company.address); // "Rīga, K. Valdemāra iela 2A"
} else {
  console.log('Uzņēmums nav atrasts.');
}
    
        
    
      
    

      
      
    
        
    SQL vaicājums
    
        
    Izmanto POST /api/sql ar JSON ķermeni { sql: "SELECT ..." } priekš
        sarežģītākiem filtriem. Tabulas nosaukums vienmēr ir resursa ID pēdiņās.
    
        
    
          ⚠️
          
    Atļauti tikai SELECT vaicājumi. INSERT, UPDATE, DELETE tiks noraidīti.
    
        
    
        
    
          
    
            TypeScript
            
          
    
          
    import type { Company } from './types';

// Resursa ID — tabulas nosaukums CKAN datubāzē
const RESOURCE_ID = '25e80bf3-f107-4ab4-89ef-251b5b9374e9';

interface SqlResponse {
  total:   number;
  records: Company[];
  fields:  Array<{ id: string; type: string }>; // kolonnu meta-dati
}

/**
 * Izpilda brīvu SQL SELECT vaicājumu pret uzņēmumu reģistru.
 *
 * @param sql - SELECT vaicājums (max 2000 rakstzīmes)
 * @returns     vaicājuma rezultāts
 */
async function runSql(sql: string): Promise<SqlResponse> {
  const response = await fetch('http://localhost:3000/api/sql', {
    method:  'POST',
    headers: { 'Content-Type': 'application/json' },
    body:    JSON.stringify({ sql }),
  });

  if (!response.ok) {
    const err = await response.json() as { error: string };
    throw new Error(err.error);
  }

  return response.json() as Promise<SqlResponse>;
}

// ── Lietošanas piemērs: aktīvie SIA Rīgā ─────────────────────────────────
const result = await runSql(`
  SELECT regcode, name, address, registered
  FROM "${RESOURCE_ID}"
  WHERE type    = 'SIA'
    AND address ILIKE '%Rīga%'
    AND closed  = ' '
  ORDER BY registered DESC
  LIMIT 20
`);

console.log(`Atrasti: ${result.total}`);
result.records.forEach(c => {
  console.log(`${c.regcode}  ${c.name}`);
});
    
        
    
      
    

      
      
    
        
    Kļūdu apstrāde
    
        
    Serveris vienmēr atgriež JSON ar error lauku, ja kas noiet greizi.
        Zemāk ir universāls palīgfunkcijas piemērs.
    
        
    
          
    
            TypeScript
            
          
    
          
    /** HTTP statusa kodi, ko serveris var atgriezt: */
// 200 – viss kārtībā
// 400 – nepareizs pieprasījums (trūkst parametru, kļūdains SQL)
// 429 – pārāk daudz pieprasījumu (rate limit — 120/min vai 20/min SQL)
// 500 – servera kļūda
// 502 – data.gov.lv nav pieejams

interface ApiError {
  error: string; // cilvēklasāms kļūdas apraksts
}

/**
 * Daudzpusīgs fetch iesaiņotājs ar kļūdu apstrādi.
 * Izmet Error ar kļūdas ziņojumu no servera, ja response nav ok.
 */
async function apiFetch<T>(
  url:    string,
  init?:  RequestInit
): Promise<T> {
  const response = await fetch(url, init);

  if (!response.ok) {
    // Serveris vienmēr atgriež { error: "..." } kļūdas gadījumā
    const body = await response.json() as ApiError;
    throw new Error(`[${response.status}] ${body.error}`);
  }

  return response.json() as Promise<T>;
}

// ── Lietošanas piemērs ar try/catch ──────────────────────────────────────
try {
  const data = await apiFetch<SearchResponse>(
    'http://localhost:3000/api/search?q=Latvijas'
  );
  console.log(data.records);
} catch (err) {
  // err.message saturēs servera kļūdas aprakstu
  console.error('Kļūda:', err);
}
