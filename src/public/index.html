export interface Company {
  _id:               number;        // iekšējais DB ID
  regcode:           string;        // reģistrācijas numurs (11 cipari)
  sepa:              string;        // SEPA maksājuma identifikators
  name:              string;        // pilns nosaukums
  name_before_quotes:string;        // nosaukuma daļa pirms pēdiņām
  name_in_quotes:    string;        // nosaukuma daļa pēdiņās
  name_after_quotes: string;        // nosaukuma daļa pēc pēdiņām
  without_quotes:    number;        // 1 ja nosaukumā nav pēdiņu
  regtype:           string;        // reģistra veida kods  (U, B, ...)
  regtype_text:      string;        // reģistra veida nosaukums
  type:              string;        // juridiskā forma (SIA, AS, ...)
  type_text:         string;        // juridiskās formas pilns nosaukums
  registered:        string;        // ISO datums — reģistrācijas datums
  terminated:        string | null; // ISO datums — izslēgšanas datums
  closed:            string;        // ' ' = aktīvs, 'L' = likvidēts, ...
  address:           string;        // juridiskā adrese
  index:             number;        // pasta indekss
  addressid:         number;        // adreses reģistra ID
  region:            number;        // reģiona kods
  city:              number;        // pilsētas / novada kods
  atvk:              number;        // ATVK teritorijas kods
  reregistration_term: string;      // pārreģistrācijas termiņš
}

/** Atbilde ko atgriež /api/search */
export interface SearchResponse {
  total:   number;    // kopējais atbilstošo ierakstu skaits datubāzē
  records: Company[]; // šajā lapā atgrieztie ieraksti
  meta: {
    resource_id:  string; // CKAN resursa ID
    limit:        number; // cik ierakstu prasīti
    offset:       number; // no kura ieraksta sākts
    upstream_url: string; // faktiskais URL uz data.gov.lv
  };
}

    import type { SearchResponse } from './types';

/**
 * Meklē uzņēmumus pēc nosaukuma vai jebkura teksta lauka.
 * @param query  meklēšanas frāze, piem. "SIA Latvijas"
 * @param limit  cik rezultātus atgriezt (max 50, noklusējums 10)
 * @param offset izlaist N ierakstus (lapošanai)
 */
async function searchByName(
  query: string,
  limit = 10,
  offset = 0,
): Promise<SearchResponse> {
  const url = new URL('http://localhost:3000/api/search');
  url.searchParams.set('q',      query);
  url.searchParams.set('limit',  String(limit));
  url.searchParams.set('offset', String(offset));

  const res = await fetch(url.toString());
  if (!res.ok) {
    const err = await res.json() as { error: string };
    throw new Error(err.error);
  }
  return res.json() as Promise<SearchResponse>;
}

// ── Piemērs ───────────────────────────────────────────────────────────────
const result = await searchByName('Latvijas Banka', 5);
console.log(`Atrasti: ${result.total}`);
result.records.forEach(c => console.log(`${c.regcode}  ${c.name}`));

    import type { SearchResponse, Company } from './types';

/**
 * Atrod uzņēmumu pēc precīza reģistrācijas numura.
 * @param regcode 11 ciparu numurs, piem. "40003009497"
 * @returns Company objekts vai null, ja netika atrasts
 */
async function findByRegcode(regcode: string): Promise<Company | null> {
  const url = new URL('http://localhost:3000/api/search');
  url.searchParams.set('regcode', regcode);

  const res = await fetch(url.toString());
  if (!res.ok) {
    const err = await res.json() as { error: string };
    throw new Error(err.error);
  }
  const data = await res.json() as SearchResponse;
  return data.records[0] ?? null;  // null ja nekas netika atrasts
}

// ── Piemērs ───────────────────────────────────────────────────────────────
const company = await findByRegcode('40003009497');
if (company) {
  console.log(company.name);    // "LATVIJAS BANKA"
  console.log(company.address);
} else {
  console.log('Uzņēmums nav atrasts.');
}

    import type { Company } from './types';

const RESOURCE_ID = '25e80bf3-f107-4ab4-89ef-251b5b9374e9';

interface SqlResponse {
  total:   number;
  records: Company[];
  fields:  Array<{ id: string; type: string }>;
}

/** Izpilda brīvu SQL SELECT vaicājumu pret uzņēmumu reģistru. */
async function runSql(sql: string): Promise<SqlResponse> {
  const res = await fetch('http://localhost:3000/api/sql', {
    method:  'POST',
    headers: { 'Content-Type': 'application/json' },
    body:    JSON.stringify({ sql }),
  });

  if (!res.ok) {
    const err = await res.json() as { error: string };
    throw new Error(err.error);
  }
  return res.json() as Promise<SqlResponse>;
}

// ── Piemērs: aktīvie SIA Rīgā ────────────────────────────────────────────
const result = await runSql(`
  SELECT regcode, name, address, registered
  FROM "${RESOURCE_ID}"
  WHERE type    = 'SIA'
    AND address ILIKE '%Rīga%'
    AND closed  = ' '
  ORDER BY registered DESC
  LIMIT 20
`);
result.records.forEach(c => console.log(`${c.regcode}  ${c.name}`));
    
        
    
      
    

      
      
    
        
    Kļūdu apstrāde
    
        
    
          Serveris vienmēr atgriež
          { error: "..." }
          kļūdas gadījumā. Universāls iesaiņotājs:
        
    
        
    
          
    
            TypeScript
            
          
    
          
    // HTTP statusi:
// 200 – viss kārtībā
// 400 – nepareizs pieprasījums (trūkst parametru, kļūdains SQL)
// 429 – pārāk daudz pieprasījumu (120/min globāli, 20/min SQL)
// 500 – servera kļūda
// 502 – data.gov.lv nav pieejams

/** Daudzpusīgs fetch iesaiņotājs ar kļūdu apstrādi. */
async function apiFetch<T>(url: string, init?: RequestInit): Promise<T> {
  const res = await fetch(url, init);

  if (!res.ok) {
    const body = await res.json() as { error: string };
    throw new Error(`[${res.status}] ${body.error}`);
  }
  return res.json() as Promise<T>;
}

// ── Piemērs ───────────────────────────────────────────────────────────────
try {
  const data = await apiFetch<SearchResponse>(
    'http://localhost:3000/api/search?q=Latvijas'
  );
  console.log(data.records);
} catch (err) {
  console.error('Kļūda:', err); // kļūdas ziņojums no servera
}
